  <script>
    document.addEventListener('DOMContentLoaded', function () {
      var form = document.getElementById('contact-form');
      if (!form) return;

      var statusEl = document.getElementById('form-status');
      var submitButton = form.querySelector('.submit-button');
      var spinner = document.getElementById('submit-spinner');
      var honeypot = form.querySelector('input[name="website"]');

      function setStatus(message, type) {
        statusEl.textContent = message;
        statusEl.className = 'form-status ' + type;
      }

      function setLoading(isLoading) {
        submitButton.disabled = isLoading;
        spinner.classList.toggle('visible', isLoading);
      }

      function clearStatus() {
        statusEl.textContent = '';
        statusEl.className = 'form-status hidden';
      }

      function validateField(field) {
        if (!field.checkValidity()) {
          if (field.validity.valueMissing) {
            return 'Please complete the ' + field.previousElementSibling.textContent.toLowerCase() + ' field.';
          }
          if (field.validity.typeMismatch) {
            return 'Please enter a valid ' + field.previousElementSibling.textContent.toLowerCase() + '.';
          }
          if (field.validity.patternMismatch) {
            return 'Please provide a valid ' + field.previousElementSibling.textContent.toLowerCase() + '.';
          }
        }
        return '';
      }

      form.addEventListener('input', function () {
        if (!statusEl.classList.contains('hidden')) {
          clearStatus();
        }
      });

      form.addEventListener('submit', function (event) {
        event.preventDefault();
        clearStatus();

        if (honeypot && honeypot.value.trim() !== '') {
          setStatus('Thank you! Your message has been sent successfully.', 'success');
          form.reset();
          return;
        }

        var requiredFields = Array.from(form.querySelectorAll('input[required], textarea[required]'));
        for (var i = 0; i < requiredFields.length; i++) {
          var error = validateField(requiredFields[i]);
          if (error) {
            setStatus(error, 'error');
            requiredFields[i].focus();
            return;
          }
        }

        setLoading(true);
        setStatus('Sending your message, please wait...', 'success');

        window.setTimeout(function () {
          setLoading(false);
          form.reset();
          setStatus('Thank you! Your message has been sent successfully.', 'success');
        }, 1400);
      });
    });
  </script>
